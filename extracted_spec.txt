import { defaultCache } from '@serwist/next/worker'
import { Serwist, NetworkFirst, CacheFirst, StaleWhileRevalidate, ExpirationPlugin, BackgroundSyncQueue } from 'serwist'

const serwist = new Serwist({
  precacheEntries: self.__SW_MANIFEST,
  skipWaiting: false,          // on demande à l'utilisateur avant de mettre à jour
  clientsClaim: true,
  navigationPreload: true,
  runtimeCaching: [
    // 1. App shell & pages : réseau d'abord, cache en secours, timeout court
    {
      matcher: ({ request }) => request.mode === 'navigate',
      handler: new NetworkFirst({
        cacheName: 'secs-pages',
        networkTimeoutSeconds: 3,
        plugins: [new ExpirationPlugin({ maxEntries: 60, maxAgeSeconds: 30 * 24 * 3600 })],
      }),
    },
    // 2. Photos produits : cache d'abord (elles ne changent presque jamais)
    {
      matcher: ({ url }) => url.pathname.includes('/storage/v1/object/public/produits'),
      handler: new CacheFirst({
        cacheName: 'secs-images-produits',
        plugins: [new ExpirationPlugin({ maxEntries: 2000, maxAgeSeconds: 180 * 24 * 3600, purgeOnQuotaError: true })],
      }),
    },
    // 3. Polices
    {
      matcher: ({ request }) => request.destination === 'font',
      handler: new CacheFirst({ cacheName: 'secs-fonts' }),
    },
    // 4. Référentiel produits (API) : SWR pour un affichage instantané puis rafraîchissement
    {
      matcher: ({ url }) => url.pathname.startsWith('/api/referentiel'),
      handler: new StaleWhileRevalidate({ cacheName: 'secs-referentiel' }),
    },
    ...defaultCache,
  ],
  fallbacks: {
    entries: [{ url: '/offline', matcher: ({ request }) => request.destination === 'document' }],
  },
})

// Background Sync : les ventes créées offline partent dès le retour du réseau
const ventesQueue = new BackgroundSyncQueue('secs-ventes-queue', { maxRetentionTime: 14 * 24 * 60 })
self.addEventListener('fetch', (event) => {
  if (event.request.method === 'POST' && new URL(event.request.url).pathname === '/api/ventes') {
    event.respondWith(
      fetch(event.request.clone()).catch(async () => {
        await ventesQueue.pushRequest({ request: event.request })
        return new Response(JSON.stringify({ queued: true }), { status: 202, headers: { 'Content-Type': 'application/json' } })
      })
    )
  }
})

5.5 Le composant d'installation cross-OS
C'est le point exact de ta demande. Le comportement diffère selon l'OS, il faut donc un composant intelligent.
'use client'
import { useEffect, useState } from 'react'

type Platform = 'android-chrome' | 'ios-safari' | 'desktop-chromium' | 'desktop-safari' | 'firefox' | 'installed' | 'unknown'

function detectPlatform(): Platform {
  if (typeof window === 'undefined') return 'unknown'
  const ua = navigator.userAgent
  const isStandalone =
    window.matchMedia('(display-mode: standalone)').matches ||
    window.matchMedia('(display-mode: window-controls-overlay)').matches ||
    (window.navigator as any).standalone === true
  if (isStandalone) return 'installed'

  const isIOS = /iPad|iPhone|iPod/.test(ua) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1)
  if (isIOS) return 'ios-safari'
  if (/Android/.test(ua)) return 'android-chrome'
  if (/Firefox/.test(ua)) return 'firefox'
  if (/Safari/.test(ua) && !/Chrome|Chromium|Edg/.test(ua)) return 'desktop-safari'
  return 'desktop-chromium'
}

export function InstallSECS() {
  const [platform, setPlatform] = useState<Platform>('unknown')
  const [prompt, setPrompt] = useState<any>(null)

  useEffect(() => {
    setPlatform(detectPlatform())
    const onPrompt = (e: Event) => { e.preventDefault(); setPrompt(e) }
    window.addEventListener('beforeinstallprompt', onPrompt)
    window.addEventListener('appinstalled', () => { setPrompt(null); setPlatform('installed') })
    return () => window.removeEventListener('beforeinstallprompt', onPrompt)
  }, [])

  if (platform === 'installed') return null

  // Cas 1 : Chrome/Edge Android, Windows, macOS, Linux, ChromeOS → installation en 1 clic
  if (prompt) {
    return (
      <button onClick={async () => {
        prompt.prompt()
        const { outcome } = await prompt.userChoice
        if (outcome === 'accepted') setPrompt(null)
      }}>
        Installer SECS sur cet appareil
      </button>
    )
  }

  // Cas 2 : iOS / iPadOS Safari → instructions manuelles illustrées (pas d'API disponible)
  if (platform === 'ios-safari') {
    return <IosInstallSheet /> // "Partager" → "Sur l'écran d'accueil" → "Ajouter"
  }

  // Cas 3 : Safari macOS 17+ → menu Fichier → Ajouter au Dock
  if (platform === 'desktop-safari') {
    return <SafariMacInstructions />
  }

  // Cas 4 : Firefox → pas d'installation PWA. On propose Chrome/Edge ou l'usage navigateur.
  if (platform === 'firefox') {
    return <FirefoxNotice />
  }

  return null
}

6.2 Le pattern Outbox (file d'attente sortante)
Règle absolue : la caisse n'attend jamais le réseau.
Encaissement validé
   ↓
1. Écriture immédiate dans Dexie (ventes + outbox)   ← instantané, ~5 ms
2. Facture PDF générée localement avec pdf-lib        ← instantané, offline
3. Impression envoyée à l'imprimante                  ← instantané
4. UI : "Vente enregistrée" + monnaie à rendre         ← instantané
   ↓ (en tâche de fond, invisible pour le caissier)
5. Le worker de sync tente l'envoi au serveur
6. Si échec → retry avec backoff exponentiel (1s, 2s, 4s... max 5 min)
7. Si succès → sync_status = 'synced', numero_serveur enregistré
Idempotence :
l'
id
(uuid v7) est généré par le client et sert de clé primaire côté serveur. Si le réseau coupe après l'écriture serveur mais avant la réponse, le retry ne crée pas de doublon (
INSERT ... ON CONFLICT (id) DO NOTHING
).
6.3 Numérotation des factures en mode offline
Problème classique : deux postes offline génèrent la facture n°43.
Solution : numérotation composite.
{PREFIXE_POSTE}-{YYYYMMDD}-{SEQUENCE_LOCALE}
Exemple : C1-20260729-0042
Chaque poste de caisse reçoit un
préfixe unique
à l'enrôlement (C1, C2, C3...)
La séquence est locale au poste et à la journée →
aucune collision possible
Le serveur peut, si l'exigence fiscale l'impose, attribuer en plus un
numéro séquentiel global
à la synchronisation, conservé en parallèle du numéro local (les deux figurent sur la facture réimprimée)

    .from('produits')
    .select('*')

Génération automatique des références par catégorie
(exigence "Référencement - Génération d'une référence par catégorie") :
CREATE OR REPLACE FUNCTION generer_reference() RETURNS trigger AS $$
DECLARE p text; n integer;
BEGIN
  IF NEW.reference IS NULL OR NEW.reference = '' THEN
    SELECT coalesce(prefixe_ref, code) INTO p FROM categories WHERE id = NEW.categorie_id;
    UPDATE categories SET compteur_ref = compteur_ref + 1 WHERE id = NEW.categorie_id RETURNING compteur_ref INTO n;
    NEW.reference := p || '-' || lpad(n::text, 4, '0');
  END IF;
  RETURN NEW;
END $$ LANGUAGE plpgsql;
CREATE TRIGGER trg_generer_reference BEFORE INSERT ON produits FOR EACH ROW EXECUTE FUNCTION generer_reference();

  type          text NOT NULL CHECK (type IN ('entree','sortie_vente','sortie_perte','sortie_retour_fournisseur','ajustement_inventaire','transfert','retour_client')),
  quantite      numeric(14,3) NOT NULL,      -- signé : + entrée, - sortie
  cout_unitaire integer,                     -- pour le calcul du PMP
  reference_doc text,                        -- n° facture, n° bon de livraison
  vente_id      uuid REFERENCES ventes(id),
  fournisseur_id uuid REFERENCES fournisseurs(id),
  motif         text,
  utilisateur_id uuid NOT NULL REFERENCES utilisateurs(id),
  cree_le       timestamptz NOT NULL DEFAULT now(),
  cree_le_client timestamptz
);
CREATE INDEX idx_mouvements_produit_date ON mouvements_stock(produit_id, cree_le DESC);

-- Le stock est TOUJOURS un SUM, jamais une valeur écrasée
CREATE OR REPLACE FUNCTION recalculer_stock_cache() RETURNS trigger AS $$
BEGIN
  UPDATE produits p
     SET stock_cache = (SELECT sum(quantite) FROM mouvements_stock WHERE produit_id = p.id),
         updated_at = now()
   WHERE p.id = coalesce(NEW.produit_id, OLD.produit_id);
  RETURN NULL;
END $$ LANGUAGE plpgsql;
CREATE TRIGGER trg_stock_cache AFTER INSERT OR UPDATE OR DELETE ON mouvements_stock
  FOR EACH ROW EXECUTE FUNCTION recalculer_stock_cache();
Calcul du coût : PMP (Prix Moyen Pondéré)
, recalculé à chaque entrée. C'est la méthode la plus adaptée à la quincaillerie (produits fongibles, prix d'achat variables). FIFO est possible mais alourdit inutilement.

CREATE MATERIALIZED VIEW mv_ventes_jour AS
SELECT date_trunc('day', cree_le_client AT TIME ZONE 'Africa/Porto-Novo')::date AS jour,
       count(*) AS nb_ventes,
       sum(total) AS ca,
       sum(marge) AS marge,
       sum(total) / nullif(count(*),0) AS panier_moyen
FROM ventes WHERE statut = 'validee'
GROUP BY 1;
CREATE UNIQUE INDEX ON mv_ventes_jour(jour);

CREATE MATERIALIZED VIEW mv_top_produits AS
SELECT lv.produit_id, p.nom, p.reference, c.nom AS categorie,
       date_trunc('month', v.cree_le_client)::date AS mois,
       sum(lv.quantite) AS qte_vendue,
       sum(lv.total_ligne) AS ca,
       sum(lv.total_ligne - (lv.cout_unitaire * lv.quantite)) AS marge
FROM lignes_vente lv
JOIN ventes v ON v.id = lv.vente_id AND v.statut = 'validee'
JOIN produits p ON p.id = lv.produit_id
JOIN categories c ON c.id = p.categorie_id
GROUP BY 1,2,3,4,5;

-- Rafraîchissement toutes les 15 minutes via pg_cron
SELECT cron.schedule('refresh_kpi', '*/15 * * * *', $$
  REFRESH MATERIALIZED VIEW CONCURRENTLY mv_ventes_jour;
  REFRESH MATERIALIZED VIEW CONCURRENTLY mv_top_produits;
$$);

-- Un caissier ne voit que ses propres ventes du jour
CREATE POLICY caissier_ses_ventes ON ventes FOR SELECT
  USING (
    caissier_id = auth.uid()
    AND cree_le_client >= current_date
    OR a_permission(auth.uid(), 'ventes.lire_toutes')
  );

-- Personne ne peut modifier une vente validée (sauf avoir)

  USING (a_permission(auth.uid(), 'ventes.annuler'));

22. ROADMAP TECHNIQUE (alignée sur les 5 phases du cahier des charges)
Nom officiel du logiciel : Obasa.
Toutes les interfaces, factures, icônes, notifications, exports et documents utilisateurs doivent afficher « Obasa ». Le terme SECS désigne uniquement le cahier des charges historique et ne doit plus apparaître dans l'expérience utilisateur.
22.0 Paramètres métier désormais validés
Les décisions suivantes sont figées pour la v1 :
TVA et e-MECeF : obligatoires.
L'intégration de normalisation fiscale est incluse dans le chemin critique de la facturation, avec gestion des files d'attente offline, des retries, des erreurs et du mode dégradé validé par la DGI.
Multi-sites et multi-postes : obligatoires.
Chaque donnée métier sensible porte
site_id
. Chaque poste reçoit un préfixe de facture unique, par exemple
CTN-C1
,
CTN-C2
ou
ABO-C1
, afin d'éviter toute collision entre points de vente.
Multi-utilisateurs simultanés en caisse : obligatoire.
Les ventes sont append-only, idempotentes et synchronisées par mouvements. Les droits et sessions de caisse sont isolés par utilisateur, poste et site.
Vente à crédit : exclue de la v1.
Aucun compte client, encours ou paiement différé ne doit être implémenté dans le premier périmètre.
Unités complexes : exclues de la v1.
La v1 utilise des unités simples et non convertibles : pièce, sac, kg, litre, mètre, botte.
Mobile Money : exclu de la v1.
Les moyens initiaux sont espèces, virement et chèque si retenu par le client.
Hébergement : Supabase Cloud.
Les migrations SQL restent versionnées dans le dépôt et la propriété du code d'intégration reste au client.
Langue : français uniquement.
Préparer les libellés dans des fichiers de traduction, sans ajouter d'interface multilingue en v1.
Devis et bons de commande : hors v1.
Réserver les identifiants et les relations nécessaires pour une phase ultérieure, sans les développer maintenant.
22.0.1 Règle e-MECeF avant toute mise en production
La mise en production est bloquée tant que le parcours suivant n'est pas validé avec les identifiants et règles applicables au contribuable :
Créer une facture locale offline avec un numéro unique par site et poste.
Synchroniser la vente dès le retour du réseau sans créer de doublon.
Transmettre la facture à e-MECeF avec journalisation du payload et de la réponse.
Enregistrer le code, le QR code et le statut de normalisation retournés.
Réimprimer ou télécharger la facture normalisée.
Rejouer automatiquement les erreurs temporaires avec backoff.
Mettre en file les erreurs définitives pour traitement par le gérant.
Produire un rapport de rapprochement entre ventes locales, ventes synchronisées et factures normalisées.
Le fonctionnement offline ne doit jamais falsifier le statut fiscal : une facture locale non encore normalisée doit être clairement marquée
« en attente de normalisation e-MECeF »
jusqu'à confirmation du serveur.

26. LES 10 RÈGLES À NE JAMAIS VIOLER
La caisse n'attend jamais le réseau.
Écriture locale d'abord, toujours.
Les montants sont des entiers.
Aucun float sur le FCFA.
Le stock est une somme de mouvements
, jamais une colonne qu'on écrase.
Les lignes de vente stockent un snapshot
du nom et du prix.
L'ID de vente est généré par le client.
C'est la clé de l'idempotence.
Aucun appel direct au back-end depuis un composant.
Tout passe par
lib/repositories
.
Pas de
skipWaiting
: on ne met jamais à jour l'app pendant un encaissement.
L'audit log est alimenté par des triggers
, jamais par le code applicatif.
Tester sur un Android d'entrée de gamme
, pas sur ta machine de dev.
Une vente ne se supprime pas.
On fait un avoir. Toujours.
